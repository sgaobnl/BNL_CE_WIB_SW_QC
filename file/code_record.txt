# Open Hardware Manager and connect to FPGA
open_hw
connect_hw_server

# Get the IBERT core
set iberts [get_hw_sio_iberts]
if {[llength $iberts] == 0} {
    puts "No IBERT cores detected. Exiting."
    exit
}

# Select the first IBERT core
set ibert [lindex $iberts 0]
open_hw_target [get_hw_targets *]

# Get all IBERT serial I/O links
set ibert_links [get_hw_sio_links -of_objects $ibert]

# Check if links exist
if {[llength $ibert_links] == 0} {
    puts "No IBERT links detected."
    close_hw
    exit
}

# Select the first available IBERT link
set link0 [lindex $ibert_links 0]

# Print all available properties for the IBERT link
report_property [get_hw_sio_links $link0]

# Open Hardware Manager and connect to FPGA
open_hw
connect_hw_server

# Get the IBERT core
set iberts [get_hw_sio_iberts]
if {[llength $iberts] == 0} {
    puts "No IBERT cores detected. Exiting."
    exit
}

# Select the first IBERT core
set ibert [lindex $iberts 0]
open_hw_target [get_hw_targets *]

# Get all RX channels from IBERT (BER is measured at RX)
set rx_channels [get_hw_sio_rxs -of_objects $ibert]

# Check if RX channels are available
if {[llength $rx_channels] == 0} {
    puts "No RX channels detected. Exiting."
    close_hw
    exit
}

puts "IBERT RX Channel BER Reports:"
puts "--------------------------------------"

# Loop through each RX channel and get BER & Bit Count
foreach rx $rx_channels {
    # Get Bit Error Rate (BER)
    set ber [get_property RX_BER $rx]

    # Get Received Bit Count
    set bit_count [get_property RX_RECEIVED_BIT_COUNT $rx]

    # Print results
    puts "RX Channel: $rx"
    puts "  Bit Error Rate (BER): $ber"
    puts "  Total Bit Count: $bit_count"
    puts "--------------------------------------"
}

# Close hardware session
close_hw
